{{- /* Responsive for front-matter images (cover images, project cards). Usage: {{ partial "responsive-image.html" (dict "src" .Params.image // "/img/foo.jpeg" or a full URL "alt" "Description of the image" "class" "img-fluid" // optional "sizes" "(min-width: 992px) 730px, 100vw" // optional "eager" true // optional: above-the-fold cover "style" "height: 315px" // optional inline style ) }} If the image lives in assets/ (e.g. assets/img/foo.jpeg for src "/img/foo.jpeg"), Hugo generates WebP versions at several widths and the browser picks the smallest one that fits. The original file is never published. Anything else (files still in static/, remote URLs, SVGs) renders as a plain , exactly as before, so images can move to assets/ one at a time. */ -}} {{- $src := .src -}} {{- $alt := .alt | default "" -}} {{- $class := .class | default "img-fluid" -}} {{- $sizes := .sizes | default "(min-width: 992px) 730px, 100vw" -}} {{- $eager := .eager | default false -}} {{- $style := .style | default "" -}} {{- $res := "" -}} {{- if and $src (not (hasPrefix $src "http")) -}} {{- with resources.Get (strings.TrimPrefix "/" $src) -}} {{- if in (slice "jpeg" "png" "webp") .MediaType.SubType -}} {{- $res = . -}} {{- end -}} {{- end -}} {{- end -}} {{- with $res -}} {{- $widths := slice -}} {{- range slice 480 800 1200 1600 -}} {{- if lt . $res.Width -}}{{- $widths = $widths | append . -}}{{- end -}} {{- end -}} {{- /* Include the original width only when it's below the largest step. */ -}} {{- if lt $res.Width 1600 -}}{{- $widths = $widths | append $res.Width -}}{{- end -}} {{- $srcset := slice -}} {{- $largest := "" -}} {{- range $widths -}} {{- $img := $res.Resize (printf "%dx webp q80" .) -}} {{- $srcset = $srcset | append (printf "%s %dw" $img.RelPermalink $img.Width) -}} {{- $largest = $img -}} {{- end -}} {{- else -}} {{- with $src -}} {{- end -}} {{- end -}}
